cmake_minimum_required(VERSION 3.18)
project(ascend_hal_sharing_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CANN_ROOT "/usr/local/Ascend/ascend-toolkit/latest" CACHE PATH "CANN toolkit installation")
set(DRIVER_ROOT "/usr/local/Ascend/driver" CACHE PATH "Installed Ascend driver")
find_path(ACL_INCLUDE_DIR acl/acl.h PATHS "${CANN_ROOT}/include" "${CANN_ROOT}/${CMAKE_SYSTEM_PROCESSOR}-linux/include" NO_DEFAULT_PATH REQUIRED)
find_path(HAL_INCLUDE_DIR ascend_hal.h PATHS "${DRIVER_ROOT}/include" "${CANN_ROOT}/include/driver" "${CANN_ROOT}/${CMAKE_SYSTEM_PROCESSOR}-linux/include/driver" NO_DEFAULT_PATH REQUIRED)
find_library(ACL_LIBRARY ascendcl PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/${CMAKE_SYSTEM_PROCESSOR}-linux/lib64" NO_DEFAULT_PATH REQUIRED)
find_library(HAL_LIBRARY ascend_hal PATHS "${DRIVER_ROOT}/lib64/driver" "${DRIVER_ROOT}/lib64" NO_DEFAULT_PATH REQUIRED)
get_filename_component(HAL_LIBRARY_DIR "${HAL_LIBRARY}" DIRECTORY)
get_filename_component(ACL_LIBRARY_DIR "${ACL_LIBRARY}" DIRECTORY)
message(STATUS "HAL headers: ${HAL_INCLUDE_DIR}")
message(STATUS "HAL library: ${HAL_LIBRARY}")
message(STATUS "ACL library: ${ACL_LIBRARY}")
foreach(target host_share device_share hal_map_probe multi_share)
    add_executable(${target} ${target}.cpp)
    target_include_directories(${target} SYSTEM PRIVATE "${HAL_INCLUDE_DIR}")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(${target} PRIVATE "${HAL_LIBRARY}" pthread)
    set_target_properties(${target} PROPERTIES BUILD_RPATH "${HAL_LIBRARY_DIR};${ACL_LIBRARY_DIR}")
endforeach()
foreach(target host_share device_share multi_share)
    target_include_directories(${target} SYSTEM PRIVATE "${ACL_INCLUDE_DIR}")
    target_link_libraries(${target} PRIVATE "${ACL_LIBRARY}")
endforeach()
target_link_libraries(multi_share PRIVATE ${CMAKE_DL_LIBS})
