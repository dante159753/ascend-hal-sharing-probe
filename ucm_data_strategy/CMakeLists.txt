cmake_minimum_required(VERSION 3.18)
project(ucm_data_strategy_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CANN_ROOT "/usr/local/Ascend/ascend-toolkit/latest" CACHE PATH "CANN toolkit installation")
set(DRIVER_ROOT "/usr/local/Ascend/driver" CACHE PATH "Installed Ascend driver")
find_path(ACL_INCLUDE_DIR acl/acl.h PATHS "${CANN_ROOT}/include" "${CANN_ROOT}/${CMAKE_SYSTEM_PROCESSOR}-linux/include" NO_DEFAULT_PATH REQUIRED)
find_path(HAL_INCLUDE_DIR ascend_hal.h PATHS "${DRIVER_ROOT}/include" "${CANN_ROOT}/include/driver" "${CANN_ROOT}/${CMAKE_SYSTEM_PROCESSOR}-linux/include/driver" NO_DEFAULT_PATH REQUIRED)
find_library(ACL_LIBRARY ascendcl PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/${CMAKE_SYSTEM_PROCESSOR}-linux/lib64" NO_DEFAULT_PATH REQUIRED)
find_library(HAL_LIBRARY ascend_hal PATHS "${DRIVER_ROOT}/lib64/driver" "${DRIVER_ROOT}/lib64" NO_DEFAULT_PATH REQUIRED)
find_package(fmt 11.2.0 CONFIG QUIET)
if(NOT fmt_FOUND)
    include(FetchContent)
    FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 11.2.0 GIT_SHALLOW TRUE)
    set(FMT_TEST OFF CACHE BOOL "" FORCE)
    set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(fmt)
endif()
find_package(Threads REQUIRED)
add_executable(data_strategy_demo main.cpp data_strategy.cc trans/device.cc trans/ascend/hal/hal_memory.cc)
target_compile_definitions(data_strategy_demo PRIVATE UCM_RUNTIME_ASCEND_HAL=1 UCM_RUNTIME_ASCEND_FAMILY=1)
target_include_directories(data_strategy_demo PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(data_strategy_demo SYSTEM PRIVATE "${ACL_INCLUDE_DIR}" "${HAL_INCLUDE_DIR}")
target_compile_options(data_strategy_demo PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(data_strategy_demo PRIVATE fmt::fmt "${ACL_LIBRARY}" "${HAL_LIBRARY}" Threads::Threads)
message(STATUS "HAL headers: ${HAL_INCLUDE_DIR}")
message(STATUS "HAL library: ${HAL_LIBRARY}")
message(STATUS "ACL library: ${ACL_LIBRARY}")
